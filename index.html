    PurchaseService.server.luau
    -- Never trust the client
local Remote = Remotes.Purchase

Remote.OnServerEvent:Connect(function(player, itemId)
    if not RateLimit:Check(player) then return end

    local item = Catalog[itemId]
    if not item then return end

    local data = Profiles:Get(player)
    if data.Coins < item.Price then return end

    data.Coins -= item.Price
    Inventory:Grant(player, itemId)
end)
